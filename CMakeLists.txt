cmake_minimum_required(VERSION 3.31)

project(parallel_course_2026 VERSION 1.0.0 LANGUAGES C CXX)

include(CTest)

if(COURSE_ENABLE_CUDA)
  enable_language(CUDA)
  set(CMAKE_CUDA_STANDARD 20)
  set(CMAKE_CUDA_STANDARD_REQUIRED ON)
  set(CMAKE_CUDA_EXTENSIONS OFF)
endif()

set(CMAKE_C_STANDARD "${COURSE_C_STANDARD}")
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
set(CMAKE_CXX_STANDARD "${COURSE_CXX_STANDARD}")
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

include(cmake/CourseDependencies.cmake)

message(STATUS "Curso: C${COURSE_C_STANDARD}, C++${COURSE_CXX_STANDARD}, OpenMP ${COURSE_OPENMP_VERSION}, MPI ${COURSE_MPI_STANDARD_VERSION}.0")
message(STATUS "Implementación MPI: ${COURSE_MPI_IMPLEMENTATION} ${COURSE_MPI_VERSION}")
message(STATUS "CUDA habilitado: ${COURSE_ENABLE_CUDA}; versión objetivo: ${COURSE_CUDA_VERSION}")
